Main Algorithm:
 
Get numPizzas                          // assume 1 to 5
total = 0
for i = 1 to numPizzas
    Get radius[i]
    Get price[i]
    valuePerInch[i] = ComputeValue(radius[i], price[i])
    total = total + price[i]
end for
 
if total >= 50.00
    deliveryCharge = 0
else
    deliveryCharge = 8.00
end if

totalBill = total + deliveryCharge

bestDeal = FindBestDeal(valuePerInch, numPizzas)

print "Total Bill: ", totalBill
print "Best Deal Pizza #: ", bestDeal
print "Price per sq inch: ", valuePerInch[bestDeal]


Subproblem - ComputeValue(radius, price):

area = 3.141592 * radius * radius
pricePerSquareInch = price / area 
return pricePerSquareInch


Subproblem - FindBestDeal(valuePerInch, numPizzas):

lowest = 1
for i = 2 to numPizzas
    if valuePerInch[i] < valuePerInch[lowest]
        lowest = i
    end if
end for
return lowest
